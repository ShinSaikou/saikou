# Written in tools/bench/ui_steps.py's language.
# The welcome page takes a tap made while it is still settling as nothing (the first attempt's run
# stayed on it), so it is waited for and given two seconds.

tap text=Next
tap text=Select a folder

tap text=${UPSTREAM_STORAGE_DIR}
tap text=USE THIS FOLDER
tap text=ALLOW
tap text=Next
# Install apps permission, through the system's own toggle. The first Grant on the page is its row.
tap text=Grant
check text=Allow from this source
tap desc=Navigate up
# Background battery usage: the one Grant left.
tap text=Grant
tap text=Allow
uncheck text=Send crash logs
uncheck text=Allow analytics
tap text=Next
tap text=Get started
wait text=Library timeout=30
